Python: explicit compensation states

The main path shows state transitions first. Provider request keys are kept because they are part of safe compensation; optional effect governance is covered in the implementation notes below.

# Each successful booking records the ID needed to undo it.
from ferricstore import WorkflowClient, transition, complete

client = WorkflowClient.from_url("ferric://127.0.0.1:6388")
saga = client.workflow(type="vacation-checkout", initial_state="charge_card")

@saga.state("charge_card")
def charge(ctx):
    tx = stripe.charge(85_000, idempotency_key=f"{ctx.id}:charge:v1")
    return transition("book_flight", values={"tx_id": tx.id})

@saga.state("book_flight")
def flight(ctx):
    try:
        seat = delta.book_seat("DL402", request_id=f"{ctx.id}:flight:v1")
        return transition("book_hotel", values={"seat_id": seat.id})
    except DeltaUpstream5xx:
        return transition("verify_flight")

@saga.state("verify_flight")
def verify_flight(ctx):
    seat = delta.get_booking(request_id=f"{ctx.id}:flight:v1")
    if seat:
        return transition("book_hotel", values={"seat_id": seat.id})
    return transition("refund_card")

@saga.state("book_hotel")
def hotel(ctx):
    try:
        room = marriott.book_suite(request_id=f"{ctx.id}:hotel:v1")
        return transition("rent_car", values={"room_id": room.id})
    except HotelSoldOutException:
        return transition("cancel_seat")

@saga.state("rent_car")
def car(ctx):
    rental = hertz.book_suv(request_id=f"{ctx.id}:car:v1")
    return complete(result={"status": "BOOKING_COMPLETE", "rental_id": rental.id})

@saga.state("cancel_seat", claim_values=["seat_id"])
def cancel_seat(ctx):
    delta.cancel_seat(ctx.value("seat_id"), request_id=f"{ctx.id}:cancel:v1")
    return transition("refund_card")

@saga.state("refund_card", claim_values=["tx_id"])
def refund_card(ctx):
    stripe.refund(ctx.value("tx_id"), idempotency_key=f"{ctx.id}:refund:v1")
    return complete(result={"status": "ROLLED_BACK"})

saga.start("trip-42", payload={"customer_id": "customer-42"})
saga.worker().run()
