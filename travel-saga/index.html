FerricStore Python SDK: Reversible Sagas

Animated excerpts omit the surrounding @ctx.effect declarations. FerricStore states and transitions below match all three scenarios; Stripe, Delta, Marriott, and Hertz calls are application-owned adapters.

# Compensations are explicit states; every provider mutation has a stable key
from ferricstore import WorkflowClient, transition, complete

client = WorkflowClient.from_url("ferric://127.0.0.1:6388")
saga = client.workflow(type="vacation-checkout", initial_state="charge_card")

@saga.state("charge_card")
def charge(ctx):
    @ctx.effect("charge", "stripe.charge", operation_digest=f"charge:{ctx.id}:v1")
    def call_stripe():
        return stripe.charge(85_000, idempotency_key=f"{ctx.id}:charge:v1")
    tx = call_stripe()
    return transition("book_flight", values={"tx_id": tx.id})

@saga.state("book_flight")
def flight(ctx):
    @ctx.effect("flight", "delta.book", operation_digest=f"flight:{ctx.id}:v1")
    def book():
        return delta.book_seat("DL402", request_id=f"{ctx.id}:flight:v1")
    try:
        seat = book()
        return transition("book_hotel", values={"seat_id": seat.id})
    except DeltaUpstream5xx:
        # A 500 is ambiguous: verify by stable request ID before retrying.
        return transition("verify_flight")

@saga.state("verify_flight")
def verify_flight(ctx):
    seat = delta.get_booking(request_id=f"{ctx.id}:flight:v1")
    if seat:
        return transition("book_hotel", values={"seat_id": seat.id})
    return transition("refund_card")

@saga.state("book_hotel")
def hotel(ctx):
    @ctx.effect("hotel", "marriott.book", operation_digest=f"hotel:{ctx.id}:v1")
    def book():
        return marriott.book_suite(request_id=f"{ctx.id}:hotel:v1")
    try:
        room = book()
        return transition("rent_car", values={"room_id": room.id})
    except HotelSoldOutException:
        return transition("cancel_seat")

@saga.state("rent_car")
def car(ctx):
    @ctx.effect("car", "hertz.book", operation_digest=f"car:{ctx.id}:v1")
    def book():
        return hertz.book_suv(request_id=f"{ctx.id}:car:v1")
    rental = book()
    return complete(result={"status": "BOOKING_COMPLETE", "rental_id": rental.id})

@saga.state("cancel_seat", claim_values=["seat_id"])
def cancel_seat(ctx):
    @ctx.effect("cancel-flight", "delta.cancel", operation_digest=f"cancel:{ctx.id}:v1")
    def cancel():
        delta.cancel_seat(ctx.value("seat_id"), request_id=f"{ctx.id}:cancel:v1")
    cancel()
    return transition("refund_card")

@saga.state("refund_card", claim_values=["tx_id"])
def refund_card(ctx):
    @ctx.effect("refund", "stripe.refund", operation_digest=f"refund:{ctx.id}:v1")
    def refund():
        stripe.refund(ctx.value("tx_id"), idempotency_key=f"{ctx.id}:refund:v1")
    refund()
    return complete(result={"status": "ROLLED_BACK"})

saga.start("trip-42", payload={"customer_id": "customer-42"})
saga.worker().run()
