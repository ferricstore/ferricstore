Python: states and an approval signal

This example keeps the workflow path visible. Provider adapters are abbreviated; production idempotency and effect guards are covered in the implementation notes below.

from ferricstore import WorkflowClient, transition, complete

client = WorkflowClient.from_url("ferric://127.0.0.1:6388")
flow = client.workflow(type="ad-campaign-agent", initial_state="research")

@flow.state("research")
def research(ctx):
    competitors = web_scraper.run(ctx.payload["campaign"])
    return transition("draft_plan", payload={"competitors": competitors})

@flow.state("draft_plan")
def draft(ctx):
    plan = llm.create_campaign(ctx.payload["competitors"])
    slack.post_approval_request(
        job_id=ctx.id, plan=plan,
        idempotency_key=f"{ctx.id}:approval:v1",
    )
    return transition("await_approval", payload={"plan": plan})

@flow.state("launch_campaign")
def launch(ctx):
    campaign = google_ads.create_campaign(
        ctx.payload["plan"], request_id=f"{ctx.id}:launch:v1",
    )
    return complete(result=campaign)

@app.post("/webhooks/slack/approve")
def slack_webhook(payload: dict):
    job_id = payload["job_id"]
    client.signal(
        job_id, signal="approved", if_state="await_approval",
        transition_to="launch_campaign", values={"approved_by": payload["user"]},
        idempotency_key=f"{job_id}:approve:{payload['action_id']}:v1",
    )
    return {"status": "resumed"}

flow.start("camp-9842", payload={"campaign": "Summer SaaS Growth Blast"})
flow.worker().run()
