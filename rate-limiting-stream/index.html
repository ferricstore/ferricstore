Python: rate-limit a workflow state

from ferricstore import WorkflowClient, transition

client = WorkflowClient.from_url("ferric://127.0.0.1:6388")
flow = client.workflow(type="shopify-webhook", initial_state="categorize")

@flow.state("categorize")
def categorize(ctx):
    limit = ctx.client.ratelimit_add(
        "rl:openai:categorize", window_ms=1_000, max=100,
    )
    if not limit.allowed:
        return transition("categorize", run_at_ms=ctx.now_ms + limit.reset_ms)

    summary = ai.categorize(ctx.payload["prompt"])
    return transition("flush", values={"summary": summary})

flow.worker().run()
