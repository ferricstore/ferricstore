FerricStore Python SDK: Stream Rate-Limiting

# Real rate-limit command + durable pipeline continuation
import time
from ferricstore import FlowClient

client = FlowClient.from_url("ferric://127.0.0.1:6388")
job = client.start_and_claim(
    "webhook-912", type="shopify-webhook-stream",
    initial_state="categorize", worker="stream-1",
    payload={"prompt": "Classify this product update"},
)

limit = client.ratelimit_add(
    "rl:openai:categorize", window_ms=1_000, max=100,
)
if not limit.allowed:
    client.transition(
        job.id,
        lease_token=job.lease_token,
        fencing_token=job.fencing_token,
        from_state=job.run_state,
        to_state="categorize",
        run_at_ms=int(time.time() * 1000) + limit.reset_ms,
    )
else:
    job, summary = client.step(
        job,
        name="categorize-product:v1",
        run=lambda: ai.categorize_once(
            operation_id=f"{job.id}:categorize:v1",
            prompt=job.payload["prompt"],
        ),
        to_state="flush",
    )
