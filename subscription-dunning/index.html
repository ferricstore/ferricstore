Clean Multi-Day Subscriptions

Animated excerpts omit the surrounding @ctx.effect declarations. The State Handlers tab matches all five stages; Durable Step focuses on the first boundary in a custom claim loop. email, sms, and stripe are application-owned adapters.

# State handlers implement the same five-stage recovery shown above
import time
from ferricstore import WorkflowClient, complete, transition

client = WorkflowClient.from_url("ferric://127.0.0.1:6388")
billing = client.workflow(type="saas-subscription", initial_state="welcome")
DAY_MS = 86_400_000

def in_days(days):
    return int(time.time() * 1000) + days * DAY_MS

def charge_subscription(workflow_id, attempt):
    # Return a codec-friendly result from the application-owned Stripe adapter.
    payment = stripe.charge(
        9900,
        idempotency_key=(
            f"{workflow_id}:invoice:2026-09:attempt-{attempt}:v1"
        ),
    )
    return {"status": payment.status, "payment_id": payment.id}

@billing.state("welcome")
def welcome(ctx):
    @ctx.effect(
        "welcome-email", "email.send",
        operation_digest=f"welcome:{ctx.id}:v1",
    )
    def send_welcome():
        return email.send_welcome(
            ctx.payload["user_id"],
            idempotency_key=f"{ctx.id}:welcome:v1",
        )

    send_welcome()
    # Persist the wait and release the worker until Day 14.
    return transition(
        "pre_billing_warning",
        run_at_ms=in_days(14),
    )

@billing.state("pre_billing_warning")
def pre_billing_warning(ctx):
    @ctx.effect(
        "warning-email", "email.send",
        operation_digest=f"warning:{ctx.id}:v1",
    )
    def send_warning():
        return email.send_warning(
            ctx.payload["user_id"],
            idempotency_key=f"{ctx.id}:warning:v1",
        )

    send_warning()
    return transition(
        "charge",
        run_at_ms=in_days(1),
    )

@billing.state("charge")
def charge(ctx):
    @ctx.effect(
        "charge-attempt-1", "stripe.charge",
        operation_digest=f"charge:{ctx.id}:attempt-1:v1",
    )
    def charge_once():
        return charge_subscription(ctx.id, attempt=1)

    attempt = charge_once()
    if attempt["status"] == "paid":
        return complete(result=attempt)

    @ctx.effect(
        "dunning-email-1", "email.send",
        operation_digest=f"dunning:{ctx.id}:attempt-1:v1",
    )
    def send_dunning():
        return email.send_dunning(
            ctx.payload["user_id"],
            idempotency_key=f"{ctx.id}:dunning-1:v1",
        )

    send_dunning()
    return transition("retry_1", run_at_ms=in_days(3))

@billing.state("retry_1")
def retry_1(ctx):
    @ctx.effect(
        "charge-attempt-2", "stripe.charge",
        operation_digest=f"charge:{ctx.id}:attempt-2:v1",
    )
    def charge_once():
        return charge_subscription(ctx.id, attempt=2)

    attempt = charge_once()
    if attempt["status"] == "paid":
        return complete(result=attempt)

    @ctx.effect(
        "card-update-sms", "sms.send",
        operation_digest=f"sms:{ctx.id}:attempt-1:v1",
    )
    def send_update_link():
        return sms.send_update_link(
            ctx.payload["user_id"],
            idempotency_key=f"{ctx.id}:sms-1:v1",
        )

    send_update_link()
    return transition("retry_2", run_at_ms=in_days(3))

@billing.state("retry_2")
def retry_2(ctx):
    @ctx.effect(
        "charge-attempt-3", "stripe.charge",
        operation_digest=f"charge:{ctx.id}:attempt-3:v1",
    )
    def charge_once():
        return charge_subscription(ctx.id, attempt=3)

    payment = charge_once()
    return complete(result=payment)

billing.start("sub-user-42", payload={"user_id": "user-42"})
billing.worker().run()
# Direct durable-step API for a custom claim loop
import time
from ferricstore import FlowClient

client = FlowClient.from_url("ferric://127.0.0.1:6388")
job = client.start_and_claim(
    "sub-user-42", type="saas-subscription",
    initial_state="welcome", worker="billing-1",
    payload={"user_id": "user-42"},
)

# The closure result and next state are committed together. The returned job
# carries the refreshed lease and fence required for the next operation.
job, _ = client.step(
    job,
    name="welcome-email:v1",
    run=lambda: email.send_welcome(
        job.payload["user_id"],
        idempotency_key=f"{job.id}:welcome:v1",
    ),
    to_state="welcome_sent",
)

# Park until Day 14; no Python process or worker lease stays alive.
client.transition(
    job.id,
    lease_token=job.lease_token,
    fencing_token=job.fencing_token,
    from_state=job.run_state,
    to_state="pre_billing_warning",
    run_at_ms=int(time.time() * 1000) + 14 * 86_400_000,
)
