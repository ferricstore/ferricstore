# FerricStore Python SDK: one handler per durable state
from ferricstore import WorkflowClient, complete, transition

client = WorkflowClient.from_url("ferric://127.0.0.1:6388")
checkout = client.workflow(type="ai-checkout", initial_state="charge")

@checkout.state("charge")
def charge(ctx):
    tx = stripe.charge(
        ctx.payload["amount"],
        idempotency_key=f"{ctx.id}:charge:v1",
    )
    return transition("summarize", values={"tx_id": tx.id})

@checkout.state("summarize")
def summarize(ctx):
    summary = llm.generate(ctx.payload["prompt"])
    return transition("deliver", values={"summary": summary})

@checkout.state("deliver")
def deliver(ctx):
    email.send_receipt(
        ctx.payload["email"], tx_id=ctx.value("tx_id"),
        idempotency_key=f"{ctx.id}:receipt:v1",
    )
    return complete(result=b"sent")

checkout.worker().run()
